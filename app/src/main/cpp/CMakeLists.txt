cmake_minimum_required(VERSION 3.22.1)
project(chronicle LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_AARCH64 OFF CACHE BOOL "" FORCE)
# Portable binaries; x86 emulator must not require AVX/FMA.
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../../vendor/llama.cpp llama EXCLUDE_FROM_ALL)
add_library(chronicle SHARED bridge.cpp)
target_include_directories(chronicle PRIVATE ../../../../vendor/llama.cpp/include)
target_link_libraries(chronicle PRIVATE llama android log)
target_link_options(chronicle PRIVATE "-Wl,-z,max-page-size=16384")
